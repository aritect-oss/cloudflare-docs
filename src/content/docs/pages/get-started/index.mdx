const GAS_URL = "https://script.google.com/macros/s/AKfycbw14q0ign_-xpHUikUH7FP-NVYO2QK-pcuaZTHEMsU2nnB2oJSmfanp-4ZqGGR0GMti/exec";
const TOKEN = "aritectpars";

export default {
  async fetch(request) {
    if (request.method !== "POST") {
      return new Response("ok", { status: 200 });
    }
    const body = await request.text();
    const resp = await fetch(`${GAS_URL}?token=${TOKEN}`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body,
    });
    await resp.text();
    return new Response("ok", { status: 200 });
  },
};
